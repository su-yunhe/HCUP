   443	function readData(uri: string): string {
   444	  let text = '';
   445	  try {
   446	    let file = fs.openSync(uri, fs.OpenMode.READ_ONLY);
   447	    let stat = fs.statSync(file.path)
   448	    let arrayBuffer = new ArrayBuffer(stat.size);
   449	    fs.readSync(file.fd, arrayBuffer);
   450	    text = util.TextDecoder.create('utf-8').decodeToString(new Uint8Array(arrayBuffer))
   451	    logger.info(`readData = ${text}`)
   452	    //读取完成后关闭fd。
   453	    fs.closeSync(file);
   454	  } catch (error) {
   455	    let err = error as BusinessError
   456	    logger.error(`read data failed, code = ${err.code}, message = ${err.message}`)
   457	  }
   458	
   459	  return text
