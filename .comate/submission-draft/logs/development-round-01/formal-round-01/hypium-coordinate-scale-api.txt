    def _scale_to_position(self, pos, area_size=None):
        if area_size is None:
            area_size = self.driver.getDisplaySize().to_tuple()
        width, height = area_size
        x, y = pos
        if 0 <= x <= 1 or 0 <= y <= 1:
            x = int(x * width)
            y = int(y * height)
        return x, y
